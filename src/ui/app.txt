  function h(tag, props) {

      Object.keys(props).forEach(function (k) {
        var v = props[k];
        if (v == null || v === false) return;
        if (k.slice(0, 2) === 'on') el.addEventListener(k.slice(2), v);
        else if (k === 'class') el.className = v;
        else if (k === 'value') { if (tag === 'input') el.defaultValue = v; el.value = v; }
        else if (k === 'checked') el.checked = !!v;
        else if (k === 'style' && typeof v === 'object') Object.assign(el.style, v);
        else el.setAttribute(k, v === true ? '' : v);
      });
    }
    var kids = Array.prototype.slice.call(arguments, 2);
    (function add(list) {
      list.forEach(function (c) {
        if (c == null || c === false) return;
        if (Array.isArray(c)) return add(c);
        el.append(c instanceof Node ? c : String(c));
      });
    })(kids);
    return el;
  }

  function ago(ts) {

    if (s < 86400) return Math.round(s / 3600) + ' h ago';
    if (s < 7 * 86400) return Math.round(s / 86400) + ' d ago';
    return new Date(ts).toLocaleDateString('en', { month: 'short', day: 'numeric' });
  }

  // Number field with thousands separators: typing 500000 shows 500,000. 500k and 2m work too.
  function numIn(value, attrs) {
    var el = h('input', Object.assign({ type: 'text', inputmode: 'numeric', autocomplete: 'off', spellcheck: 'false' }, attrs || {}));
    function digits(v) {
      var t = String(v == null ? '' : v).trim();
      var m = t.match(/^([\d,]*\.?\d*)\s*([km])$/i);
      if (m && m[1]) return String(Math.round(Number(m[1].replace(/,/g, '')) * (m[2].toLowerCase() === 'k' ? 1e3 : 1e6)));
      return t.replace(/[^0-9]/g, '').replace(/^0+(?=\d)/, '').slice(0, 15);
    }
    function show(v) { var d = digits(v); return d ? NF0.format(Number(d)) : ''; }
    el.value = el.defaultValue = show(value);
    el.addEventListener('input', function () {
      var raw = el.value;
      var pos = el.selectionStart == null ? raw.length : el.selectionStart;
      var right = /[km]\s*$/i.test(raw) ? 0 : raw.slice(pos).replace(/[^0-9]/g, '').length;
      el.value = show(raw);
      var i = el.value.length;
      var seen = 0;
      while (i > 0 && seen < right) { i--; if (/\d/.test(el.value.charAt(i))) seen++; }
      try { el.setSelectionRange(i, i); } catch (e) { /* not focused */ }
    });
    Object.defineProperty(el, 'num', { get: function () { var d = digits(el.value); return d ? Number(d) : 0; } });
    return el;
  }

  // Numbers count up on the first view of a page.
  function countUp(el, to, suffix) {
    suffix = suffix || '';
    if (REDUCED || !to) { el.textContent = fmt(to) + suffix; return el; }
    var t0 = performance.now();
    function step(t) {
      var k = Math.min(1, (t - t0) / 900);
      var e = 1 - Math.pow(1 - k, 3);
      el.textContent = fmt(Math.round(to * e)) + suffix;
      if (k < 1) requestAnimationFrame(step);
    }
    el.textContent = '0' + suffix;
    requestAnimationFrame(step);
    return el;
  }

  function api(method, path, body) {
    var opts = { method: method, headers: { authorization: 'Bearer ' + KEY } };
    if (body !== undefined) { opts.headers['content-type'] = 'application/json'; opts.body = JSON.stringify(body); }
    return fetch('/admin/' + path, opts).then(function (r) {
      return r.text().then(function (t) {
        var j;
        try { j = JSON.parse(t); } catch (e) { j = { raw: t }; }
        if (r.status === 401) { logout(j.error && j.error.message); throw new Error('unauthorized'); }
        if (!r.ok) throw new Error((j.error && j.error.message) || ('HTTP ' + r.status));
        return j;
      });
    });
  }

  // Single-series bar chart, today in orange. Hover or focus shows the day's numbers.
  function barChart(series, field, first) {
    var val = function (s) { return field === 'req' ? s.req : s.p + s.c; };
    var max = Math.max.apply(null, series.map(val).concat([1]));
    var step = niceStep(max / 3);
    var top = step * 3;
    var plot = h('div', { class: 'plot' });
    var yax = h('div', { class: 'yax' });
    for (var i = 0; i <= 3; i++) {
      var v = step * i;
      var pos = (v / top) * 100;
      yax.append(h('span', { style: { bottom: pos + '%' } }, fmt(v)));
      if (i) plot.append(h('div', { class: 'grid-line', style: { bottom: pos + '%' } }));
    }
    var last = series.length - 1;
    var bars = h('div', { class: 'bars' + (first ? ' grow' : '') }, series.map(function (s, idx) {
      var v2 = val(s);
      var label = dayLabel(s.day) + ': ' + fmt(v2) + (field === 'req' ? ' requests' : ' tokens');
      var b = h('div', { class: 'bar' + (idx === last ? ' today' : '') + (v2 ? '' : ' zero'), tabindex: '0', role: 'img', 'aria-label': label, style: { '--i': String(idx) } },
        h('i', { style: { height: (v2 ? Math.max(4, (v2 / top) * 100) : 0) + '%' } }));
      var nodes = function () { return [h('b', null, dayLabel(s.day)), h('br'), fmt(s.p + s.c) + ' tokens', h('br'), fmt(s.p) + ' in · ' + fmt(s.c) + ' out', h('br'), plural(s.req, 'request')]; };
      b.addEventListener('mouseenter', function (e) { tipShow(e, nodes()); });
      b.addEventListener('mousemove', tipMove);
      b.addEventListener('mouseleave', tipHide);
      b.addEventListener('focus', function () { var r = b.getBoundingClientRect(); tipShow({ clientX: r.left, clientY: r.top + 40 }, nodes()); });
      b.addEventListener('blur', tipHide);
      return b;
    }));
    plot.append(bars);
    return h('div', null,
      h('div', { class: 'chart' }, yax, plot),
      h('div', { class: 'xax', 'aria-hidden': 'true' }, series.map(function (s, idx) { return h('span', null, idx === last ? 'Today' : idx % 2 === last % 2 ? dayLabel(s.day, true) : ''); })));
  }
  function niceStep(x) {
    if (x <= 1) return 1;
    var p = Math.pow(10, Math.floor(Math.log10(x)));
    var n = x / p;
    return (n <= 1 ? 1 : n <= 2 ? 2 : n <= 2.5 ? 2.5 : n <= 5 ? 5 : 10) * p;
  }

    function knownUrl(n) {

      var up = below < mh + 16 && r.top > below;

        docsIn.value = '';
        keyIn.value = '';

      }, fail);
    }

  // Provider health: one cell per day, colored by how many calls worked. Rate limits are not failures.
  function healthTone(x) {
    if (!x || !(x.ok + x.fail)) return 'none';
    var r = x.ok / (x.ok + x.fail);
    return r >= 0.98 ? 'good' : r >= 0.8 ? 'warn' : 'bad';
  }

  function backupCard(head) {
    var file = h('input', { type: 'file', accept: 'application/json,.json', class: 'hide', 'aria-label': 'Backup file' });
    file.addEventListener('change', function () {
      var f = file.files && file.files[0];
      if (!f) return;
      var rd = new FileReader();
      rd.onload = function () {
        var data;
        try { data = JSON.parse(rd.result); } catch (e) { toast('That file is not JSON', true); return; }
        var n = data && data.providers ? Object.keys(data.providers).length : 0;
        ask({ title: 'Restore this backup?', text: plural(n, 'provider') + ', routes, settings and agent keys are merged in. Nothing is deleted, and providers you already have keep their keys. New providers still need a key.', ok: 'Restore' }).then(function (yes) {
          if (!yes) return;
          api('POST', 'restore', { backup: data }).then(function (r) {
            CFG = r.config;
            toast('Restored ' + plural(r.counts.providers, 'provider') + ', ' + plural(r.counts.agentKeys, 'agent key'));
            loadAll().then(render);
          }, fail);
        });
        file.value = '';
      };
      rd.readAsText(f);
    });
    var dlB = btn('Download backup', function () {
      busy(dlB, 'Preparing', api('GET', 'backup')).then(function (b) {
        var url = URL.createObjectURL(new Blob([JSON.stringify(b, null, 2)], { type: 'application/json' }));
        var a = h('a', { href: url, download: 'keyrouter-backup-' + new Date().toISOString().slice(0, 10) + '.json' });
        document.body.append(a);
        a.click();
        a.remove();
        setTimeout(function () { URL.revokeObjectURL(url); }, 4000);
      }, fail);
    }, 'soft sm', 'download');
    return h('section', { class: 'card mt' },
      head('Backup', 'Move your setup to another deployment, or keep a copy', 'download'),
      h('div', { class: 'stack', style: { gap: '14px' } },
        h('div', { class: 'hint' }, 'The file has providers, limits, routes, settings, score overrides and agent keys (as hashes, so agents keep working). Provider keys and alert tokens are left out: add those again after a restore.'),
        h('div', { class: 'row' }, dlB, btn('Restore from file', function () { file.click(); }, 'soft sm', 'upload'), file)));
  }
