  function h(tag, props) {
    var el = document.createElement(tag);
    if (props) {
      Object.keys(props).forEach(function (k) {
        var v = props[k];
        if (v == null || v === false) return;
        if (k.slice(0, 2) === 'on') el.addEventListener(k.slice(2), v);
        else if (k === 'class') el.className = v;
        else if (k === 'value') { if (tag === 'input') el.defaultValue = v; el.value = v; }
        else if (k === 'checked') el.checked = !!v;
        else if (k === 'style' && typeof v === 'object') Object.assign(el.style, v);
        else el.setAttribute(k, v === true ? '' : v);
      });
    }
    var kids = Array.prototype.slice.call(arguments, 2);
    (function add(list) {
      list.forEach(function (c) {
        if (c == null || c === false) return;
        if (Array.isArray(c)) return add(c);
        el.append(c instanceof Node ? c : String(c));
      });
    })(kids);
    return el;
  }

  function whenIn(ts) {

  function ago(ts) {
    if (!ts) return 'never';
    var s = Math.round((Date.now() - ts) / 1000);
    if (s < 45) return 'just now';
    if (s < 3600) return Math.round(s / 60) + ' min ago';
    if (s < 86400) return Math.round(s / 3600) + ' h ago';
    if (s < 7 * 86400) return Math.round(s / 86400) + ' d ago';
    return new Date(ts).toLocaleDateString('en', { month: 'short', day: 'numeric' });
  }
  function dur(s) {
    if (s < 60) return s + ' s';

    return (s / 3600).toFixed(1) + ' h';
  }

  function sel(options, value, onchange, attrs) {
    attrs = attrs || {};

    Object.defineProperty(wrap, 'value', { get: function () { return cur; }, set: function (v) { cur = v; txt.textContent = labelOf(v); } });

    var items = [];
    var active = 0;
    function setActive(i) {
      active = (i + items.length) % items.length;
      items.forEach(function (it, k) { it.classList.toggle('act', k === active); });
      // Scroll only inside the menu: scrolling the page would close it.
      var it = items[active];
      if (menu && it) {
        if (it.offsetTop < menu.scrollTop) menu.scrollTop = it.offsetTop - 6;
        else if (it.offsetTop + it.offsetHeight > menu.scrollTop + menu.clientHeight) menu.scrollTop = it.offsetTop + it.offsetHeight - menu.clientHeight + 6;
      }

    function choose(v) {

      wrap.value = v;
      close();
      b.focus();
      if (changed && onchange) onchange(v);
    }
    function key(e) {
      if (e.key === 'Escape') { e.preventDefault(); e.stopPropagation(); close(); b.focus(); }
      else if (e.key === 'ArrowDown') { e.preventDefault(); setActive(active + 1); }
      else if (e.key === 'ArrowUp') { e.preventDefault(); setActive(active - 1); }
      else if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); e.stopPropagation(); choose(options[active][0]); }
      else if (e.key === 'Tab') close();
    }
    function outside(e) { if (menu && !menu.contains(e.target) && !wrap.contains(e.target)) close(); }
    // The page scrolled: keep the menu under its button, close it once the button is off screen.
    function onScroll(e) {
      if (!menu || menu.contains(e.target)) return;
      var r = b.getBoundingClientRect();
      if (r.bottom < 0 || r.top > window.innerHeight) close();
      else place();

      var up = below < mh + 16 && r.top > below;
      menu.style.left = Math.max(12, Math.min(r.left, window.innerWidth - menu.offsetWidth - 12)) + 'px';
      menu.style.top = (up ? r.top - mh - 6 : r.bottom + 6) + 'px';
      menu.classList.toggle('up', up);

      window.addEventListener('scroll', onScroll, true);
      window.addEventListener('resize', close);
      DD_CLOSE = close;
    }

      if (!menu) return;
      var m = menu;
      menu = null;
      if (DD_CLOSE === close) DD_CLOSE = null;
      m.classList.remove('on');
      setTimeout(function () { m.remove(); }, 180);
      b.setAttribute('aria-expanded', 'false');

      document.removeEventListener('keydown', key, true);
      document.removeEventListener('mousedown', outside, true);
      window.removeEventListener('scroll', onScroll, true);
      window.removeEventListener('resize', close);
    }
    b.addEventListener('click', function () { if (menu) close(); else open(); });
    b.addEventListener('keydown', function (e) { if (!menu && (e.key === 'ArrowDown' || e.key === 'ArrowUp')) { e.preventDefault(); open(); } });

  // Number field with thousands separators: typing 500000 shows 500,000. 500k and 2m work too.
  function numIn(value, attrs) {
    var el = h('input', Object.assign({ type: 'text', inputmode: 'numeric', autocomplete: 'off', spellcheck: 'false' }, attrs || {}));
    function digits(v) {
      var t = String(v == null ? '' : v).trim();
      var m = t.match(/^([\d,]*\.?\d*)\s*([km])$/i);
      if (m && m[1]) return String(Math.round(Number(m[1].replace(/,/g, '')) * (m[2].toLowerCase() === 'k' ? 1e3 : 1e6)));
      return t.replace(/[^0-9]/g, '').replace(/^0+(?=\d)/, '').slice(0, 15);
    }
    function show(v) { var d = digits(v); return d ? NF0.format(Number(d)) : ''; }
    el.value = el.defaultValue = show(value);
    el.addEventListener('input', function () {
      var raw = el.value;
      var pos = el.selectionStart == null ? raw.length : el.selectionStart;
      var right = /[km]\s*$/i.test(raw) ? 0 : raw.slice(pos).replace(/[^0-9]/g, '').length;
      el.value = show(raw);
      var i = el.value.length;
      var seen = 0;
      while (i > 0 && seen < right) { i--; if (/\d/.test(el.value.charAt(i))) seen++; }
      try { el.setSelectionRange(i, i); } catch (e) { /* not focused */ }
    });
    Object.defineProperty(el, 'num', { get: function () { var d = digits(el.value); return d ? Number(d) : 0; } });
    return el;
  }

      onchange(v);

    requestAnimationFrame(function () { place(true); });

  // Numbers count up on the first view of a page.
  function countUp(el, to, suffix) {
    suffix = suffix || '';
    if (REDUCED || !to) { el.textContent = fmt(to) + suffix; return el; }
    var t0 = performance.now();
    function step(t) {
      var k = Math.min(1, (t - t0) / 900);
      var e = 1 - Math.pow(1 - k, 3);
      el.textContent = fmt(Math.round(to * e)) + suffix;
      if (k < 1) requestAnimationFrame(step);
    }
    el.textContent = '0' + suffix;
    requestAnimationFrame(step);
    return el;
  }

  // ---------- API ----------
  function api(method, path, body) {
    var opts = { method: method, headers: { authorization: 'Bearer ' + KEY } };
    if (body !== undefined) { opts.headers['content-type'] = 'application/json'; opts.body = JSON.stringify(body); }
    return fetch('/admin/' + path, opts).then(function (r) {
      return r.text().then(function (t) {
        var j;
        try { j = JSON.parse(t); } catch (e) { j = { raw: t }; }
        if (r.status === 401) { logout(j.error && j.error.message); throw new Error('unauthorized'); }
        if (!r.ok) throw new Error((j.error && j.error.message) || ('HTTP ' + r.status));
        return j;
      });
    });
  }

  // Single-series bar chart, today in orange. Hover or focus shows the day's numbers.
  function barChart(series, field, first) {
    var val = function (s) { return field === 'req' ? s.req : s.p + s.c; };
    var max = Math.max.apply(null, series.map(val).concat([1]));
    var step = niceStep(max / 3);
    var top = step * 3;
    var plot = h('div', { class: 'plot' });
    var yax = h('div', { class: 'yax' });
    for (var i = 0; i <= 3; i++) {
      var v = step * i;
      var pos = (v / top) * 100;
      yax.append(h('span', { style: { bottom: pos + '%' } }, fmt(v)));
      if (i) plot.append(h('div', { class: 'grid-line', style: { bottom: pos + '%' } }));
    }
    var last = series.length - 1;
    var bars = h('div', { class: 'bars' + (first ? ' grow' : '') }, series.map(function (s, idx) {
      var v2 = val(s);
      var label = dayLabel(s.day) + ': ' + fmt(v2) + (field === 'req' ? ' requests' : ' tokens');
      var b = h('div', { class: 'bar' + (idx === last ? ' today' : '') + (v2 ? '' : ' zero'), tabindex: '0', role: 'img', 'aria-label': label, style: { '--i': String(idx) } },
        h('i', { style: { height: (v2 ? Math.max(4, (v2 / top) * 100) : 0) + '%' } }));
      var nodes = function () { return [h('b', null, dayLabel(s.day)), h('br'), fmt(s.p + s.c) + ' tokens', h('br'), fmt(s.p) + ' in · ' + fmt(s.c) + ' out', h('br'), plural(s.req, 'request')]; };
      b.addEventListener('mouseenter', function (e) { tipShow(e, nodes()); });
      b.addEventListener('mousemove', tipMove);
      b.addEventListener('mouseleave', tipHide);
      b.addEventListener('focus', function () { var r = b.getBoundingClientRect(); tipShow({ clientX: r.left, clientY: r.top + 40 }, nodes()); });
      b.addEventListener('blur', tipHide);
      return b;
    }));
    plot.append(bars);
    return h('div', null,
      h('div', { class: 'chart' }, yax, plot),
      h('div', { class: 'xax', 'aria-hidden': 'true' }, series.map(function (s, idx) { return h('span', null, idx === last ? 'Today' : idx % 2 === last % 2 ? dayLabel(s.day, true) : ''); })));
  }
  function niceStep(x) {
    if (x <= 1) return 1;
    var p = Math.pow(10, Math.floor(Math.log10(x)));
    var n = x / p;
    return (n <= 1 ? 1 : n <= 2 ? 2 : n <= 2.5 ? 2.5 : n <= 5 ? 5 : 10) * p;
  }

        h('div', { class: 'top-line' }, h('b', null, r.name), h('span', { class: 'muted num' }, fmt(v) + ' · ' + Math.round((v / total) * 100) + '%')),
        h('div', { class: 'track' }, h('i', { style: { width: Math.max(2, (v / max) * 100) + '%' } })));
    }));
  }

      h('div', { class: 'bill-row' }, h('span', { class: 'lbl' }, 'Pricing'),
        seg([['per-model', 'Auto'], ['free', 'All free'], ['paid', 'All paid']], bill, function (v) { bill = v; }, 'Pricing'),
        h('span', { class: 'hint' }, 'Auto reads each model\'s price. A model with a listed price is never treated as free.')))));

    function typed() { return nameIn.value.trim().toLowerCase(); }
    // Base URL KeyRouter already knows for this name, if any.
    function knownUrl(n) {
      if (CFG.presets[n]) return CFG.presets[n].baseUrl;
      var p = CFG.providers[n];
      return p && p.baseUrl ? p.baseUrl : null;
    }
    function hasKeys(n) { var p = CFG.providers[n]; return !!(p && p.keys && p.keys.length); }

    function onKey() {
      var v = keyIn.value.trim();
      var d = v ? detectProvider(v) : null;
      if (d && (!typed() || auto)) { nameIn.value = d; auto = true; }
      else if (!d && auto) { nameIn.value = ''; auto = false; }
      update();
    }

    function update() {
      var n = typed();
      var fixed = n ? knownUrl(n) : null;
      if (fixed) { urlIn.value = fixed; urlIn.readOnly = true; }
      else if (urlIn.readOnly) { urlIn.value = ''; urlIn.readOnly = false; }
      var isNew = !!n && !fixed;
      billRow.classList.toggle('open', isNew);
      var pre = CFG.presets[n];
      var keyless = !!(pre && pre.keyless);
      var v = keyIn.value.trim();
      var nameBad = !!n && !NAME_OK.test(n);
      var urlOk = /^https?:\/\/\S+$/.test(urlIn.value.trim());
      connectB.disabled = !n || nameBad || (!v && !keyless) || (isNew && !urlOk);
      var msg = null;
      var tone = '';
      if (nameBad) { msg = 'Use lowercase letters, numbers, - or _ for the name (no spaces).'; tone = 'bad'; }
      else if (!n) msg = v ? 'KeyRouter can\'t tell the provider from this key. Pick it from the list or type a new name.' : 'Paste a key. Most providers are detected from the key, or pick one from the list.';
      else if (fixed) {
        msg = (auto ? 'Detected ' + cap(n) + ' from the key. ' : '') +
          (hasKeys(n) ? 'Adds another key to ' + cap(n) + '.' : keyless && !v ? 'No key needed: press Connect.' : (pre && pre.note) || 'Base URL filled in for you.');
        tone = 'ok';
      } else msg = 'New provider "' + n + '". Add its OpenAI-compatible base URL, usually ending in /v1.';
      info.className = 'cinfo ' + tone;
      info.replaceChildren(icon(tone === 'bad' ? 'alert' : tone === 'ok' ? 'check' : 'info'), h('span', null, msg));
    }

    // ---- provider list under the name field
    var menu = null;
    var opts = [];
    var active = -1;
    function choices(showAll) {
      var q = typed();
      var all = Object.keys(CFG.presets).concat(Object.keys(CFG.providers).filter(function (n) { return !CFG.presets[n]; }));
      var exact = all.indexOf(q) >= 0;
      var list = showAll || auto || !q || exact ? all : all.filter(function (n) { return n.indexOf(q) >= 0; });
      var out = list.map(function (n) { return { name: n }; });
      if (q && !exact && NAME_OK.test(q)) out.push({ name: q, isNew: true });
      return out;
    }
    function optEl(o, i) {
      var pre = CFG.presets[o.name];
      var tags = [];
      if (o.isNew) tags.push(h('span', { class: 'tagmini' }, 'new'));
      else {
        if (hasKeys(o.name)) tags.push(h('span', { class: 'tagmini' }, 'connected'));
        if (pre && pre.keyless) tags.push(h('span', { class: 'tagmini' }, 'no key'));
        if (pre && pre.billing === 'paid') tags.push(h('span', { class: 'tagmini paid' }, 'paid'));
      }
      var on = o.name === typed();
      return h('div', { class: 'dd-opt' + (on ? ' on' : '') + (o.isNew ? ' newopt' : ''), role: 'option', 'aria-selected': String(on),
        onmousedown: function (e) { e.preventDefault(); }, onmousemove: function () { if (active !== i) setActive(i); }, onclick: function () { pick(o.name); } },
        h('span', { class: 'opt-name' }, o.isNew ? h('span', null, icon('plus'), 'Add "' + o.name + '"') : o.name),
        h('span', { class: 'opt-tags' }, tags));
    }
    function setActive(i) {
      if (!opts.length) return;
      active = (i + opts.length) % opts.length;
      var items = menu ? menu.children : [];
      for (var k = 0; k < items.length; k++) items[k].classList.toggle('act', k === active);
      var it = items[active];
      if (menu && it) {
        if (it.offsetTop < menu.scrollTop) menu.scrollTop = it.offsetTop - 6;
        else if (it.offsetTop + it.offsetHeight > menu.scrollTop + menu.clientHeight) menu.scrollTop = it.offsetTop + it.offsetHeight - menu.clientHeight + 6;
      }
    }
    function place() {
      if (!menu) return;
      var r = nameIn.getBoundingClientRect();
      if (r.bottom < 0 || r.top > window.innerHeight) { closeMenu(); return; }
      menu.style.minWidth = Math.max(r.width, 240) + 'px';
      var mh = menu.offsetHeight;
      var below = window.innerHeight - r.bottom;
      var up = below < mh + 16 && r.top > below;
      menu.style.left = Math.max(12, Math.min(r.left, window.innerWidth - menu.offsetWidth - 12)) + 'px';
      menu.style.top = (up ? r.top - mh - 6 : r.bottom + 6) + 'px';
      menu.classList.toggle('up', up);
    }
    function onScroll(e) { if (menu && !menu.contains(e.target)) place(); }
    function openMenu(showAll) {
      opts = choices(showAll);
      if (!opts.length) { closeMenu(); return; }
      var fresh = !menu;
      if (fresh) {
        if (DD_CLOSE) DD_CLOSE();
        menu = h('div', { class: 'dd-menu combo-menu', role: 'listbox', 'aria-label': 'Providers' });
        document.body.append(menu);
        window.addEventListener('scroll', onScroll, true);
        window.addEventListener('resize', closeMenu);
        DD_CLOSE = closeMenu;
        combo.classList.add('open');
        nameIn.setAttribute('aria-expanded', 'true');
      }
      menu.replaceChildren.apply(menu, opts.map(optEl));
      place();
      var cur = opts.map(function (o) { return o.name; }).indexOf(typed());
      setActive(cur >= 0 ? cur : 0);
      if (fresh) requestAnimationFrame(function () { if (menu) menu.classList.add('on'); });
    }
    function closeMenu() {
      if (!menu) return;
      var m = menu;
      menu = null;
      active = -1;
      if (DD_CLOSE === closeMenu) DD_CLOSE = null;
      m.classList.remove('on');
      setTimeout(function () { m.remove(); }, 180);
      window.removeEventListener('scroll', onScroll, true);
      window.removeEventListener('resize', closeMenu);
      combo.classList.remove('open');
      nameIn.setAttribute('aria-expanded', 'false');
    }
    function menuKey(e) {
      if (e.key === 'ArrowDown' || e.key === 'ArrowUp') {
        e.preventDefault();
        if (!menu) openMenu(true);
        else setActive(active + (e.key === 'ArrowDown' ? 1 : -1));
      } else if (e.key === 'Enter') {
        e.preventDefault();
        if (menu && active >= 0 && opts[active] && opts[active].name !== typed()) pick(opts[active].name);
        else { closeMenu(); connect(); }
      } else if (e.key === 'Escape' && menu) { e.preventDefault(); e.stopPropagation(); closeMenu(); }
      else if (e.key === 'Tab') closeMenu();
    }
    function pick(n) {
      nameIn.value = n;
      auto = false;
      closeMenu();
      update();
      var pre = CFG.presets[n];
      if (!keyIn.value.trim() && !(pre && pre.keyless)) keyIn.focus();
      else if (!urlIn.readOnly && !urlIn.value.trim()) urlIn.focus();
      else connectB.focus();
    }

    function connect() {
      var n = typed();
      var key = keyIn.value.trim();
      if (connectB.disabled) {
        if (!key && !(CFG.presets[n] && CFG.presets[n].keyless)) keyIn.focus();
        else if (!n || !NAME_OK.test(n)) nameIn.focus();
        else urlIn.focus();
        return;
      }
      var first = Promise.resolve();
      if (!knownUrl(n)) first = api('PUT', 'providers/' + encodeURIComponent(n), { baseUrl: urlIn.value.trim(), billing: bill });
      busy(connectB, key ? 'Checking key' : 'Turning on', first.then(function () {
        return api('POST', 'providers/' + encodeURIComponent(n) + '/keys', key ? { key: key, label: '' } : { keyless: true });
      }).then(function (c) { CFG = c; return api('GET', 'dashboard'); })).then(function (d) {
        DASH = d;
        keyIn.value = '';
        nameIn.value = '';
        urlIn.value = '';
        auto = false;
        var p = d.providers.filter(function (x) { return x.name === n; })[0];
        toast(p ? cap(n) + ': ' + p.statusText : 'Key added', p && (p.status === 'error' || p.status === 'warn'));
        NEWLY = n;
        render();
      }, fail);
    }

    update();
    return h('section', { class: 'card connect' },
      h('div', { class: 'card-head' }, h('div', null, h('h3', null, 'Connect a provider'), h('div', { class: 'sub' }, 'Your key stays on Cloudflare. Agents only ever get an agent key.'))),
      h('div', { class: 'stack', style: { gap: '12px' } },
        h('div', { class: 'cgrid' },
          field('API key', keyIn, null, 'ckey'),
          h('div', { class: 'field' }, h('span', null, 'Provider'), combo),
          field('Base URL', urlIn),
          connectB),
        info, billRow,
        h('div', { class: 'cfoot' },
          h('div', { class: 'getkeys' }, 'Free keys:', ['gemini', 'groq', 'openrouter', 'cerebras', 'mistral'].map(function (n) {
            return h('a', { href: KEY_PAGES[n], target: '_blank', rel: 'noopener noreferrer' }, n, icon('arrow'));
          })),
          keylessRow())));
  }

  // Providers that work without any key: one click to turn on.
  function keylessRow() {
    var names = Object.keys(CFG.presets).filter(function (n) { return CFG.presets[n].keyless && !(CFG.providers[n] && CFG.providers[n].keys.length); });
    if (!names.length) return null;
    return h('div', { class: 'keyless' },
      h('span', { class: 'hint' }, 'No key needed:'),
      names.map(function (n) {
        var b = btn(cap(n), function () {
          busy(b, 'Turning on', api('POST', 'providers/' + encodeURIComponent(n) + '/keys', { keyless: true }).then(function () { return loadAll(); })).then(function () {
            var np = DASH.providers.filter(function (x) { return x.name === n; })[0];
            toast(cap(n) + ': ' + (np ? np.statusText : 'on'), np && (np.status === 'error' || np.status === 'warn'));
            NEWLY = n;
            render();
          }, fail);
        }, 'soft sm', 'plus', { title: CFG.presets[n].note || '' });
        return b;
      }));
  }

    var keyIn = h('input', { type: 'password', placeholder: 'Paste another ' + cap(p.name) + ' key', autocomplete: 'off', 'aria-label': 'Another API key for ' + p.name, onkeydown: function (e) { if (e.key === 'Enter') addKey(); } });
    var addB = btn('Add key', addKey, 'soft', 'plus');

      if (!keyIn.value.trim()) { keyIn.focus(); return; }
      busy(addB, 'Checking', api('POST', 'providers/' + encodeURIComponent(p.name) + '/keys', { key: keyIn.value.trim(), label: '' }).then(function () { return loadAll(); })).then(function () {
        var np = DASH.providers.filter(function (x) { return x.name === p.name; })[0];
        toast('Key added. ' + (np ? np.statusText : ''));
        keyIn.value = '';
        render();

  function keyBox(p, k) {
    var ck = k.check;
    var meters = [];
    if (k.rpm) meters.push(meter('Requests this minute', k.lastMin, k.rpm));
    meters.push(meter('Requests today', k.today, k.rpd || 0, 'lime'));
    (k.rows || []).forEach(function (r) {
      var name = cap(r.what.replace(/-/g, ' ')) + ' left';
      var left = r.remaining != null ? r.remaining : null;
      var pct = r.limit && left != null ? Math.round((left / r.limit) * 100) : null;
      meters.push(h('div', null,
        h('div', { class: 'meter-lbl' }, h('span', null, name), h('span', null, h('b', null, left != null ? fmt(left) : '?'), r.limit ? ' of ' + fmt(r.limit) : '')),
        pct != null ? h('div', { class: 'track ' + (left <= 0 ? 'red' : pct <= 20 ? '' : 'lime'), role: 'meter', 'aria-label': name, 'aria-valuenow': String(left), 'aria-valuemax': String(r.limit) }, h('i', { style: { width: pct + '%' } })) : null,
        h('div', { class: 'hint', style: { marginTop: '6px' } }, 'Reported by provider' + (r.reset ? ', resets in ' + r.reset : ''))));
    });
    var ac = k.acct;
    var acctLine = ac ? h('div', { class: 'hint' },
      ac.limit != null ? 'Credits used ' + fmt(Math.round(ac.usage * 100) / 100) + ' of ' + fmt(ac.limit) + '. ' : 'Credits used ' + (Math.round((ac.usage || 0) * 100) / 100) + '. ',
      ac.freeTier ? 'Free tier: about 50 free-model requests a day.' : 'Has credits: about 1,000 free-model requests a day.') : null;
    var busyModels = (k.models || []).filter(function (m) { return m.blockedFor > 0; });
    return h('div', { class: 'keybox' },
      h('div', { class: 'kh' },
        h('span', { class: 'mono' }, k.mask), k.label ? h('span', { class: 'faint' }, k.label) : null,
        h('div', { class: 'badges' },
          ck ? (ck.ok ? h('span', { class: 'pill ready' }, icon('check'), 'Key works') : h('span', { class: 'pill error', title: ck.error }, icon('x'), ck.error)) : null,
          k.blockedFor > 0 ? h('span', { class: 'pill limited', title: k.reason }, icon('clock'), 'Paused ' + dur(k.blockedFor)) : null,
          iconBtn('trash', 'Remove key ' + k.mask, function () {
            ask({ title: 'Remove this key?', text: 'Key ' + k.mask + ' will be removed from ' + cap(p.name) + '. Agents keep working on your other keys.', ok: 'Remove key', danger: true }).then(function (yes) {
              if (yes) act(api('DELETE', 'providers/' + encodeURIComponent(p.name) + '/keys/' + k.id), 'Key removed').then(reload, function () {});
            });
          }, 'sm danger'))),
      h('div', { class: 'meters' }, meters),
      acctLine,
      k.reason && k.blockedFor > 0 ? h('div', { class: 'hint' }, 'Reason: ' + k.reason) : null,
      busyModels.length ? h('div', { class: 'hint' }, 'Paused models: ', busyModels.map(function (m) { return m.model + ' (' + dur(m.blockedFor) + ', ' + m.reason + ')'; }).join(', ')) : null,
      !k.rpm && !k.rpd && !(k.rows || []).length ? h('div', { class: 'hint' }, 'No limits reported yet. KeyRouter learns them from the first rate limit it hits.') : null);
  }

  // Sheet with one provider's models, free and paid in separate tabs.
  function openModels(name, which) {
    var last = document.activeElement;
    var scrim = h('div', { class: 'scrim' });
    var body = h('div', { class: 'sheet-body' }, h('div', { class: 'empty' }, h('span', { class: 'spin' }), ' Loading models'));
    var head = h('div', { class: 'sheet-head' });
    var sheet = h('div', { class: 'sheet', role: 'dialog', 'aria-modal': 'true', 'aria-label': name + ' models' }, head, body);
    function close() {
      sheet.classList.remove('on');
      scrim.classList.remove('on');
      document.removeEventListener('keydown', esc);
      setTimeout(function () { sheet.remove(); scrim.remove(); if (last && last.focus) last.focus(); }, 460);
    }
    function esc(e) { if (e.key === 'Escape') close(); }
    scrim.addEventListener('click', close);
    document.addEventListener('keydown', esc);
    document.body.append(scrim, sheet);
    requestAnimationFrame(function () { scrim.classList.add('on'); sheet.classList.add('on'); });
    var q = '';
    var search = h('input', { type: 'text', placeholder: 'Search models', 'aria-label': 'Search models', style: { flex: '1', minWidth: '160px' }, oninput: function () { q = search.value.toLowerCase(); list(); } });
    var models = [];
    function list() {
      var rows = models.filter(function (m) { return (which === 'free' ? m.free : !m.free) && (!q || m.id.toLowerCase().indexOf(q) >= 0); })
        .sort(function (a, b) { return b.score - a.score; });
      body.replaceChildren(rows.length ? h('div', { class: 'card', style: { padding: '6px 22px' } }, h('div', { class: 'list' }, rows.slice(0, 400).map(modelRow))) :
        h('div', { class: 'empty' }, which === 'free' ? 'No free models here.' : 'No paid models here.'));
    }
    function modelRow(m) {
      var key = name + '/' + m.id;
      return h('div', { class: 'li' },
        h('div', { style: { minWidth: '0', flex: '1' } },
          h('div', { class: 't mono', style: { overflow: 'hidden', textOverflow: 'ellipsis', whiteSpace: 'nowrap' } }, m.id),
          h('div', { class: 's' }, m.ctx ? fmt(m.ctx) + ' context · ' : '', m.cats.filter(function (c) { return c !== 'general'; }).join(', ') || 'general', m.tools === false ? ' · no tools' : '')),
        h('span', { class: 'chip num', title: 'Strength score' }, 'Score ' + m.score),
        sw(!m.off, function (on) { api('PUT', 'overrides', { key: key, off: !on }).then(function (c) { CAT = c; toast(on ? 'Model on' : 'Model off: never used'); }, fail); }, 'Use ' + m.id));
    }
    function draw() {
      var free = models.filter(function (m) { return m.free; }).length;
      head.replaceChildren(
        h('div', { class: 'row', style: { justifyContent: 'space-between' } }, h('h2', null, name + ' models'), iconBtn('x', 'Close', close, 'ghost')),
        h('div', { class: 'row' }, seg([['free', 'Free', free], ['paid', 'Paid', models.length - free]], which, function (v) { which = v; list(); }, 'Price'), search));
      list();
    }
    api('GET', 'catalog').then(function (c) {
      CAT = c;
      models = c.models.filter(function (m) { return m.provider === name; });
      draw();
    }, function (e) { body.replaceChildren(h('div', { class: 'empty err' }, e.message)); });
  }

    var ta = h('textarea', { rows: '12', spellcheck: 'false', 'aria-label': 'Custom routes JSON' }, JSON.stringify(CFG.routes, null, 2));
    var errBox = h('div', { class: 'err', style: { fontSize: '14px' } });
    var routesAcc = accordion('Edit custom routes (JSON)', h('div', { class: 'stack' },
      h('div', { class: 'hint' }, 'Example: { "fast": [ { "provider": "groq", "model": "llama-3.3-70b-versatile" } ] }'),
      ta, errBox,
      h('div', { class: 'row' },
        btn('Save routes', function () {
          var routes;
          try { routes = JSON.parse(ta.value); } catch (e) { errBox.textContent = 'JSON error: ' + e.message; return; }
          errBox.textContent = '';
          act(api('PUT', 'routes', { routes: routes }), 'Routes saved').then(function () { ta.value = JSON.stringify(CFG.routes, null, 2); }, function (e) { errBox.textContent = e.message; });
        }, 'sm'),
        btn('Undo changes', function () { ta.value = JSON.stringify(CFG.routes, null, 2); errBox.textContent = ''; }, 'soft sm'))));
    body.append(
      h('section', { class: 'card' },
        h('div', { class: 'card-head' }, h('div', null, h('h3', null, 'Routing'), h('div', { class: 'sub' }, 'Which models "auto" may use'))),
        h('div', { class: 'stack', style: { gap: '18px' } },
          h('div', { class: 'stack', style: { gap: '8px' } }, seg([['free', 'Free only'], ['mixed', 'Free first, then paid'], ['best', 'Strongest first']], mode, function (v) { mode = v; hint.textContent = MODE_HINT[v]; }, 'Routing mode'), hint),
          h('div', { class: 'form-grid' },
            field('Paid requests per day', capIn, 'Empty means no limit. Paid today: ' + fmt(DASH.paidToday) + '.'),
            field('Models to try per request', att),
            field('Timeout per try (seconds)', to),
            field('Wait for a free slot (seconds)', waitIn, 'When every model is briefly rate limited, hold the call this long instead of failing. 0 turns it off.')),
          h('div', null, btn('Save settings', function () {
            act(api('PUT', 'settings', { mode: mode, paidDailyCap: capIn.num, maxAttempts: Number(att.value), timeoutMs: Number(to.value) * 1000, waitSecs: waitIn.value.trim() === '' ? undefined : Number(waitIn.value) }), 'Settings saved').then(render, function () {});
          }, 'orange')))),
      h('section', { class: 'card mt' },
        h('div', { class: 'card-head', style: { marginBottom: '8px' } }, h('div', null, h('h3', null, 'Custom routes'), h('div', { class: 'sub' }, 'Optional. Smart names like auto and coding need none.'))),
        routesAcc.button, routesAcc.box));
  }

  window.addEventListener('hashchange', function () {
    var p = location.hash.slice(1);
    if (KEY && CFG && p && p !== PAGE) { PAGE = p; render(); }
  });
  window.addEventListener('scroll', tipHide, { passive: true });

  if (KEY) boot();
  else renderLogin();
})();
